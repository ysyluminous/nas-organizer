# 数据库设计

使用 SQLite，数据库文件默认 `data/nas_organizer.db`。

## 表结构

### files — 文件索引

```sql
CREATE TABLE files (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    path            TEXT    NOT NULL UNIQUE,      -- 完整路径
    volume          TEXT    NOT NULL,             -- 所属卷 volume1..5
    filename        TEXT    NOT NULL,
    ext             TEXT,                         -- 扩展名（小写）
    size            INTEGER NOT NULL,             -- 字节
    mtime           REAL    NOT NULL,             -- 修改时间 unix
    atime           REAL    NOT NULL,             -- 访问时间 unix
    ctime           REAL    NOT NULL,
    head_hash       TEXT,                         -- 头 64KB SHA256
    tail_hash       TEXT,                         -- 尾 64KB SHA256
    full_hash       TEXT,                         -- 全量 SHA256
    ftype           TEXT,                         -- 音乐/视频/图书/文档/...
    domain          TEXT,                         -- 个人/工作/生活/资源
    temp            TEXT,                         -- 热/温/冷
    confidence      REAL DEFAULT 1.0,             -- 分类置信度
    classified_by   TEXT,                         -- path_rule/ext/keyword/model/manual
    is_duplicate    INTEGER DEFAULT 0,            -- 是否重复组中的非主文件
    duplicate_of    INTEGER,                      -- 指向主文件 id
    status          TEXT DEFAULT 'active',        -- active/moved/deleted
    scanned_at      REAL    NOT NULL,
    FOREIGN KEY (duplicate_of) REFERENCES files(id)
);

CREATE INDEX idx_files_path    ON files(path);
CREATE INDEX idx_files_size    ON files(size);
CREATE INDEX idx_files_fullhash ON files(full_hash);
CREATE INDEX idx_files_ftype   ON files(ftype);
CREATE INDEX idx_files_domain  ON files(domain);
CREATE INDEX idx_files_temp    ON files(temp);
CREATE INDEX idx_files_mtime   ON files(mtime);
```

### tags / file_tags — 标签

```sql
CREATE TABLE tags (
    id    INTEGER PRIMARY KEY AUTOINCREMENT,
    name  TEXT NOT NULL UNIQUE,
    color TEXT
);

CREATE TABLE file_tags (
    file_id INTEGER NOT NULL,
    tag_id  INTEGER NOT NULL,
    PRIMARY KEY (file_id, tag_id),
    FOREIGN KEY (file_id) REFERENCES files(id),
    FOREIGN KEY (tag_id)  REFERENCES tags(id)
);
```

### plans — 迁移计划单

```sql
CREATE TABLE plans (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    name        TEXT NOT NULL,
    description TEXT,
    created_at  REAL NOT NULL,
    status      TEXT NOT NULL DEFAULT 'draft',  -- draft/confirmed/executing/done/rolled_back
    item_count  INTEGER DEFAULT 0
);
```

### plan_items — 计划明细

```sql
CREATE TABLE plan_items (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    plan_id     INTEGER NOT NULL,
    file_id     INTEGER NOT NULL,
    action      TEXT NOT NULL,   -- move/copy/delete/rename
    src_path    TEXT NOT NULL,
    dst_path    TEXT,
    reason      TEXT,            -- 为什么这么移动
    status      TEXT DEFAULT 'pending', -- pending/confirmed/done/failed/skipped
    error       TEXT,
    FOREIGN KEY (plan_id) REFERENCES plans(id),
    FOREIGN KEY (file_id) REFERENCES files(id)
);

CREATE INDEX idx_plan_items_plan ON plan_items(plan_id);
CREATE INDEX idx_plan_items_status ON plan_items(status);
```

### op_logs — 操作日志（用于回滚）

```sql
CREATE TABLE op_logs (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    plan_item_id INTEGER NOT NULL,
    action      TEXT NOT NULL,
    src_path    TEXT NOT NULL,
    dst_path    TEXT NOT NULL,
    sha256      TEXT,
    size        INTEGER,
    executed_at REAL NOT NULL,
    rolled_back INTEGER DEFAULT 0,
    FOREIGN KEY (plan_item_id) REFERENCES plan_items(id)
);
```

### scan_state — 扫描状态

```sql
CREATE TABLE scan_state (
    volume      TEXT PRIMARY KEY,
    last_scan_at REAL NOT NULL,
    file_count  INTEGER NOT NULL,
    status      TEXT NOT NULL  -- running/done/failed
);
```

## 常用查询示例

### 统计各卷文件数与占用

```sql
SELECT volume, COUNT(*) AS files, SUM(size)/1024/1024/1024 AS gb
FROM files WHERE status='active'
GROUP BY volume;
```

### 查找重复组（同全量哈希）

```sql
SELECT full_hash, COUNT(*) AS n, SUM(size) AS wasted
FROM files
WHERE full_hash IS NOT NULL AND status='active'
GROUP BY full_hash HAVING n > 1
ORDER BY wasted DESC;
```

### 待整理的冷数据大文件

```sql
SELECT path, size/1024/1024/1024 AS gb
FROM files
WHERE temp='冷' AND size > 2*1024*1024*1024 AND status='active'
ORDER BY size DESC LIMIT 50;
```

### 分类置信度低的疑难件

```sql
SELECT path, ftype, domain, confidence
FROM files
WHERE confidence < 0.8 AND status='active'
ORDER BY confidence;
```
